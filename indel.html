        function spawnBalloon() {

            if (isGameOver || isPaused) return;

            let availablePool = balloonTypes.filter(b => {

                if (b.requiredLevel) return level >= b.requiredLevel;

                return true;

            });


            let rand = Math.random() * 100;

            let cumulative = 0;

            let selectedType = availablePool[0];

            for (let b of availablePool) {

                cumulative += (b.chance || 4);

                if (rand <= cumulative) {

                    selectedType = b;

                    break;

                }

            }


            createBalloonElement(selectedType);

        }

            const handlePopAction = (e, isPetOrAuto = false) => {

                if (e && e.stopPropagation) e.stopPropagation();

                if (e && e.preventDefault) e.preventDefault();

                if (isGameOver || isPaused) return;


                const clientX = e && e.clientX ? e.clientX : (balloon.getBoundingClientRect().left + 40);

                const clientY = e && e.clientY ? e.clientY : (balloon.getBoundingClientRect().top + 40);


                if (!isPetOrAuto) {

                    triggerThrowingHandAndProjectile(clientX, clientY);

                } else {

                    triggerSilentProjectile(clientX, clientY);

                }

                if (currentHp > 0) {

                }

                if (leakInterval) clearInterval(leakInterval);

                popBalloon(balloon, selectedType, clientX, clientY);

            };

            balloon._popByPet = (clientX, clientY) => handlePopAction({ clientX, clientY }, true);


            layer.appendChild(balloon);

            if (skillChronoRiftLevel > 0) {

                speedMultiplier *= (1 - (skillChronoRiftLevel * 0.12));

            }


            const duration = (Math.random() * 4 + 8) / Math.max(0.2, speedMultiplier); 

            

            const animation = balloon.animate([

                { transform: `translateY(0px)` },

                { transform: `translateY(-${arena.clientHeight + 180}px)` }

                duration: duration * 1000,

                easing: 'linear'

            });


            const checkPauseInterval = setInterval(() => {

                if (isGameOver || !balloon.parentNode) {

                    clearInterval(checkPauseInterval);

                }

                    animation.pause();

                } else {

                    animation.play();

                }

            }, 100);


            animation.onfinish = () => {

                if (voidInterval) clearInterval(voidInterval);

                if (leakInterval) clearInterval(leakInterval);

        function triggerThrowingHandAndProjectile(targetX, targetY) {

            const startY = arenaRect.height - 35;

            const endY = targetY - arenaRect.top;


            const dx = endX - startX;

            const dy = endY - startY;

            const angleRad = Math.atan2(dy, dx);

            hand.classList.remove('hand-throwing');

            hand.classList.add('hand-throwing');

            proj.style.top = `${startY - 35}px`;

            proj.style.setProperty('--dx', `${dx}px`);

            proj.style.setProperty('--dy', `${dy}px`);

            proj.style.setProperty('--rot', `${angleDeg + 45}deg`);

        function buyPet(id) {

            const pet = pets.find(p => p.id === id);

            if (coins >= pet.price) {

                coins -= pet.price;

                pet.unlocked = true;

                if (activePetIds.length < 2) {

                    activePetIds.push(pet.id);

                }

                playSound('coin');

                updateHUD();

                renderShop();

            }

        }


        function startAdForPet(id) {

            pendingAdPetId = id;

            const btnContainer = document.getElementById('adButtonsContainer');

            btnContainer.classList.add('hidden');

            

            let timeLeft = 3;

            timerDisplay.textContent = `Ad playing... ${timeLeft}s`;

                timeLeft--;

                    btnContainer.classList.remove('hidden');

        function buyUpgrade(id) {

            const upg = upgrades.find(u => u.id === id);

            if (upg.maxLvl && upg.level >= upg.maxLvl) return;


            if (coins >= upg.cost) {

                coins -= upg.cost;

                upg.level++;

                upg.cost = Math.round(upg.cost * upg.costMult);

                

                if (id === 'damage') projectileDamage++;

                if (id === 'coin') coinMultiplier++;

                if (id === 'autopilot') autoPopperLevel++;

                if (id === 'skill_lightning') skillChainLightningLevel = upg.level;

                if (id === 'skill_vortex') skillGravityVortexLevel = upg.level;

                if (id === 'skill_sonic') skillSonicBoomLevel = upg.level;

                if (id === 'skill_chrono') skillChronoRiftLevel = upg.level;


                playSound('coin');

                updateHUD();

                renderShop();

            }

        }
